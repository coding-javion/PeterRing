    \draw[thick, line width=1pt, arrowed] (V2) -- (1.25, 1.3) node[left=1pt, font=\small] {$n$};
    \draw[thick, line width=1pt, arrowed] (1.75, 1.3) node[right=1pt, font=\small] {$j$} -- (V2);
  \end{scope}

  % Minus sign
  \node[font=\large] at (-1.8, 0.4) {$-$};

  % ========================================================

  % ========================================================
  \begin{scope}[shift={(0.2, 0)}]
    \coordinate (V1) at (0, 0);

    % Dashed interaction line
    \draw[thick, dashed, line width=1pt] (V1) -- (V2);

    \draw[thick, line width=1pt, arrowed] (V1) -- (-0.25, 1.3) node[left=1pt, font=\small] {$m$};

  \node[font=\large] at (2.8, 0.4) {$=$};
