\documentclass[tikz,border=10pt]{standalone}
\usepackage{fontspec}
\usepackage[mathrm=sym]{unicode-math}
\setmainfont{texgyrepagella-regular.otf}[
    BoldFont = texgyrepagella-bold.otf,
    ItalicFont = texgyrepagella-italic.otf,
    BoldItalicFont = texgyrepagella-bolditalic.otf
]
\setmathfont{texgyrepagella-math.otf}[AutoFakeBold = 2.5]
\usepackage{tikz}
\usetikzlibrary{arrows.meta, calc, decorations.pathmorphing}

\begin{document}
\begin{tikzpicture}[>=Stealth, line cap=round, line join=round]

  % Dimensions of plot box
  \def\W{7.0}
  \def\H{10.6}

  % Outer rectangular frame
  \draw[line width=1.1pt] (0, 0) rectangle (\W, \H);

  % Energy scale: y = 5.6 + E * 0.95 cm
  \def\yzero{5.6}
  \def\yscale{0.95}

  % Y-axis ticks and labels (-4 to +4 MeV in steps of 1)
  \foreach \E in {-4, -3, -2, -1, 0, 1, 2, 3, 4} {
    \pgfmathsetmacro{\ypos}{\yzero + \E * \yscale}
    \draw[line width=0.8pt] (0, \ypos) -- (0.15, \ypos);
    \draw[line width=0.8pt] (\W, \ypos) -- (\W - 0.15, \ypos);
    \node[left=3pt, font=\normalsize] at (0, \ypos) {\E};
  }

  % Y-axis label
  \node[font=\large, align=center] at (-0.6, 10.0) {$E$\\[-2pt]\normalsize(MeV)};

  % Nucleus label
  \node[font=\LARGE\bfseries] at (6.2, 9.2) {$^{18}\mathrm{O}$};

  % Helper macro for level drawing:
  \newcommand{\level}[4]{%
    \pgfmathsetmacro{\y}{\yzero + #3 * \yscale}
    \draw[line width=1.1pt] (#1, \y) -- (#2, \y) node[right=1.5pt, font=\small] {$#4$};%
  }

  % ==================== COLUMN 1: Exp ====================
  \def\xAL{1.1} \def\xAR{2.1}
  \level{\xAL}{\xAR}{-3.92}{0^+}
  \level{\xAL}{\xAR}{-1.94}{2^+}

  % Group near 0 MeV
  \pgfmathsetmacro{\yAfour}{\yzero - 0.38 * \yscale}
  \pgfmathsetmacro{\yAzero}{\yzero - 0.28 * \yscale}
  \pgfmathsetmacro{\yAtwo}{\yzero + 0.05 * \yscale}
  \draw[line width=1.1pt] (\xAL, \yAfour) -- (\xAR, \yAfour);
  \draw[line width=1.1pt] (\xAL, \yAzero) -- (\xAR, \yAzero);
  \draw[line width=1.1pt] (\xAL, \yAtwo) -- (\xAR, \yAtwo);
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xAR, \yAtwo) {$2^+$};
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xAR, \yAzero) {$0^+$};
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xAR + 0.22, \yAfour - 0.03) {$4^+$};

  % Triplet at ~1.3-1.5: 2+, 0+, 3+
  \pgfmathsetmacro{\yA}{\yzero + 1.28 * \yscale}
  \pgfmathsetmacro{\yB}{\yzero + 1.38 * \yscale}
  \pgfmathsetmacro{\yC}{\yzero + 1.48 * \yscale}
  \draw[line width=1.1pt] (\xAL, \yA) -- (\xAR, \yA);
  \draw[line width=1.1pt] (\xAL, \yB) -- (\xAR, \yB);
  \draw[line width=1.1pt] (\xAL, \yC) -- (\xAR, \yC);
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xAR, \yB) {$0^+$};
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xAR + 0.42, \yC + 0.04) {$3^+$};
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xAR + 0.42, \yA - 0.04) {$2^+$};

  \level{\xAL}{\xAR}{3.25}{4^+}

  % ==================== COLUMN 2: G ====================
  \def\xBL{3.0} \def\xBR{4.0}
  \level{\xBL}{\xBR}{-2.70}{0^+}
  \level{\xBL}{\xBR}{-1.58}{2^+}
  \level{\xBL}{\xBR}{-0.66}{4^+}

  % Closely spaced pair at -0.20 and -0.10: 0+ and 2+
  \pgfmathsetmacro{\yBA}{\yzero - 0.20 * \yscale}
  \pgfmathsetmacro{\yBB}{\yzero - 0.10 * \yscale}
  \draw[line width=1.1pt] (\xBL, \yBA) -- (\xBR, \yBA);
  \draw[line width=1.1pt] (\xBL, \yBB) -- (\xBR, \yBB);
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xBR, \yBB + 0.02) {$2^+$};
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xBR + 0.20, \yBA - 0.02) {$0^+$};

  \level{\xBL}{\xBR}{0.54}{3^+}
  \level{\xBL}{\xBR}{3.52}{4^+}

  % ==================== COLUMN 3: G + polarization ====================
  \def\xCL{4.9} \def\xCR{5.9}
  \level{\xCL}{\xCR}{-4.30}{0^+}
  \level{\xCL}{\xCR}{-1.93}{2^+}

  % Closely spaced pair at -0.30 and -0.10: 4+ and 0+
  \pgfmathsetmacro{\yCA}{\yzero - 0.30 * \yscale}
  \pgfmathsetmacro{\yCB}{\yzero - 0.10 * \yscale}
  \draw[line width=1.1pt] (\xCL, \yCA) -- (\xCR, \yCA);
  \draw[line width=1.1pt] (\xCL, \yCB) -- (\xCR, \yCB);
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xCR, \yCB + 0.02) {$0^+$};
  \node[anchor=west, font=\footnotesize, inner sep=1.5pt] at (\xCR, \yCA - 0.02) {$4^+$};

  \level{\xCL}{\xCR}{0.20}{2^+}
  \level{\xCL}{\xCR}{1.08}{3^+}
  \level{\xCL}{\xCR}{4.22}{4^+}

  % ==================== BOTTOM LABELS ====================
  % Col 1: Exp
  \node[font=\large\bfseries] at (1.6, 0.6) {Exp};

  % Col 2: Particle lines + wavy interaction line
  \draw[line width=0.9pt] (3.15, 0.35) -- (3.15, 0.85);
  \draw[line width=0.9pt] (3.85, 0.35) -- (3.85, 0.85);
  \draw[line width=0.9pt, decorate, decoration={snake, amplitude=1.2pt, segment length=4pt}] (3.15, 0.6) -- (3.85, 0.6);

  % Col 3: Bare G + 3p-1h polarization diagram
  \draw[line width=0.9pt] (4.75, 0.35) -- (4.75, 0.85);
  \draw[line width=0.9pt] (5.35, 0.35) -- (5.35, 0.85);
  \draw[line width=0.9pt, decorate, decoration={snake, amplitude=1.2pt, segment length=4pt}] (4.75, 0.6) -- (5.35, 0.6);

  \node[font=\normalsize] at (5.5, 0.6) {$+$};

  \draw[line width=0.9pt] (5.65, 0.35) -- (5.65, 0.85);
  \draw[line width=0.9pt] (6.65, 0.35) -- (6.65, 0.85);
  \draw[line width=0.9pt, decorate, decoration={snake, amplitude=1.2pt, segment length=3.5pt}] (5.65, 0.6) -- (5.95, 0.6);
  \draw[line width=0.9pt] (6.15, 0.6) circle (0.2);
  \draw[line width=0.9pt, decorate, decoration={snake, amplitude=1.2pt, segment length=3.5pt}] (6.35, 0.6) -- (6.65, 0.6);

\end{tikzpicture}
\end{document}
