\documentclass[tikz,border=6pt]{standalone}
\usepackage{fontspec}
\usepackage[mathrm=sym]{unicode-math}
\setmainfont{texgyrepagella-regular.otf}[
    BoldFont = texgyrepagella-bold.otf,
    ItalicFont = texgyrepagella-italic.otf,
    BoldItalicFont = texgyrepagella-bolditalic.otf
]
\setmathfont{texgyrepagella-math.otf}[AutoFakeBold = 2.5]
\usepackage{tikz}
\usetikzlibrary{arrows.meta, decorations.markings, decorations.pathmorphing}

\begin{document}
\begin{tikzpicture}[>=Stealth, line cap=round, line join=round,
  arrowed/.style={
    decoration={
      markings,
      mark=at position 0.55 with {\arrow{Stealth[length=4.8pt, width=3.3pt]}}
    },
    postaction={decorate}
  },
  arrowed_double/.style={
    decoration={
      markings,
      mark=at position 0.55 with {\arrow{Stealth[length=5.5pt, width=4.8pt]}}
    },
    postaction={decorate}
  },
  phonon/.style={
    decorate,
    decoration={snake, amplitude=1.4pt, segment length=4.5pt}
  }
]

  % Left diagram: particle-phonon coupling
  \begin{scope}[shift={(-2.2, 0)}]
    \coordinate (V) at (0, 0);

    % Incoming particle from bottom
    \draw[thick, line width=1pt, arrowed] (0, -1.3) node[right=1pt, font=\small] {$n$} -- (V);

    % Outgoing particle to up-left
    \draw[thick, line width=1pt, arrowed] (V) -- (-0.75, 1.3) node[left=1pt, font=\small] {$m$};

    % Outgoing phonon wavy line to up-right
    \draw[thick, line width=1pt, phonon] (V) -- (0.75, 1.3) node[right=1pt, font=\small] {$\nu$};
  \end{scope}

  % Right diagram: particle-pairing vibration coupling
  \begin{scope}[shift={(2.2, 0)}]
    \coordinate (V) at (0, 0);

    % Incoming particle from bottom
    \draw[thick, line width=1pt, arrowed] (0, -1.3) node[right=1pt, font=\small] {$n$} -- (V);

    % Incoming particle from up-left
    \draw[thick, line width=1pt, arrowed] (-0.75, 1.3) node[left=1pt, font=\small] {$j$} -- (V);

    % Outgoing pairing mode (double line) to up-right
    \draw[thick, line width=0.9pt, double, double distance=1.6pt, arrowed_double] 
      (V) -- (0.75, 1.3) node[right=2pt, font=\small] {$\tau$};
  \end{scope}

\end{tikzpicture}
\end{document}
