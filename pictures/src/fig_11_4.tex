\documentclass[tikz,border=8pt]{standalone}
\usepackage{fontspec}
\usepackage[mathrm=sym]{unicode-math}
\setmainfont{texgyrepagella-regular.otf}[
    BoldFont = texgyrepagella-bold.otf,
    ItalicFont = texgyrepagella-italic.otf,
    BoldItalicFont = texgyrepagella-bolditalic.otf
]
\setmathfont{texgyrepagella-math.otf}[AutoFakeBold = 2.5]
\usepackage{tikz}
\usetikzlibrary{arrows.meta, calc}

\begin{document}
\begin{tikzpicture}[>=Stealth, line cap=round, line join=round]

  % Styles
  \tikzset{
    axis/.style={line width=0.9pt, ->},
    lvl/.style={line width=1.3pt},
    conn/.style={line width=0.8pt, dashed, dash pattern=on 3.5pt off 2.5pt},
    lbl/.style={font=\small, inner sep=1.5pt}
  }

  % ========================================================
  % LEFT SUBPLOT: 170Hf Rotational Spectrum
  % ========================================================
  % y: 1000 keV = 2.5 cm, 2000 keV = 5.0 cm, 100 keV = 0.25 cm
  \def\yk{0.0025}
  
  % Axes
  \draw[axis] (0, 0) -- (0, 5.8) node[above right=-2pt, font=\normalsize] {$E_I\ [\text{keV}]$};
  \draw[axis] (-0.2, 0) -- (5.2, 0) node[below=2pt, font=\normalsize] {$I$};

  % Y Ticks (pointing inside: to the right)
  \foreach \yval/\ylbl in {100/100, 1000/1000, 2000/2000} {
    \draw[line width=0.8pt] (0, {\yval*\yk}) -- (0.12, {\yval*\yk});
    \node[left=3pt, lbl] at (0, {\yval*\yk}) {\ylbl};
  }

  % X Ticks & Labels (pointing inside: upwards)
  \def\dx{0.65}
  \foreach \i/\spin/\xpos in {
    0/{0^+}/{0.25},
    2/{2^+}/{0.25 + 1*\dx},
    4/{4^+}/{0.25 + 2*\dx},
    6/{6^+}/{0.25 + 3*\dx},
    8/{8^+}/{0.25 + 4*\dx},
    10/{10^+}/{0.25 + 5*\dx},
    12/{12^+}/{0.25 + 6*\dx}
  } {
    \draw[line width=0.8pt] (\xpos, 0) -- (\xpos, 0.12);
    \node[below=3pt, lbl] at (\xpos, 0) {$\spin$};
  }

  % Level bars (half-width hw = 0.18)
  \def\hw{0.18}
  \coordinate (H0)  at (0.25, {0*\yk});
  \coordinate (H2)  at ({0.25 + 1*\dx}, {97.1*\yk});
  \coordinate (H4)  at ({0.25 + 2*\dx}, {308.7*\yk});
  \coordinate (H6)  at ({0.25 + 3*\dx}, {615.5*\yk});
  \coordinate (H8)  at ({0.25 + 4*\dx}, {1005.8*\yk});
  \coordinate (H10) at ({0.25 + 5*\dx}, {1450.5*\yk});
  \coordinate (H12) at ({0.25 + 6*\dx}, {1965.0*\yk});

  \foreach \pt in {H0, H2, H4, H6, H8, H10, H12} {
    \draw[lvl] ($(\pt) - (\hw, 0)$) -- ($(\pt) + (\hw, 0)$);
  }

  % Dashed rotational curve
  \draw[conn, smooth] plot coordinates {(H0) (H2) (H4) (H6) (H8) (H10) (H12)};

  % Nucleus label
  \node[font=\large] at (0.9, 4.4) {\textbf{$^{170}$Hf}};

  % ========================================================
  % RIGHT SUBPLOT: Sn Pairing Rotational Spectrum
  % ========================================================
  % Origin for right subplot
  \def\xR{6.2}
  % y: tick 0 is at y = 0.45 cm, each 1 MeV = 0.48 cm
  \def\yzero{0.45}
  \def\ym{0.48}

  % Axes
  \draw[axis] (\xR, 0) -- (\xR, 5.8) node[above right=-2pt, font=\normalsize] {$E\ [\text{MeV}]$};
  \draw[axis] ({\xR - 0.2}, 0) -- ({\xR + 6.8}, 0) node[below=2pt, font=\normalsize] {$A - A_0$};

  % Y Ticks (pointing inside: to the right)
  \foreach \yval in {0, 1, 2, 3, 4, 5, 10} {
    \draw[line width=0.8pt] (\xR, {\yzero + \yval*\ym}) -- ({\xR + 0.12}, {\yzero + \yval*\ym});
    \node[left=3pt, lbl] at (\xR, {\yzero + \yval*\ym}) {\yval};
  }

  % X Ticks & Labels (pointing inside: upwards)
  \def\dxR{0.62}
  \foreach \idx/\N in {
    0/60, 1/62, 2/64, 3/66, 4/68, 5/70, 6/72, 7/74, 8/76
  } {
    \def\curX{{\xR + 0.6 + \idx*\dxR}}
    \draw[line width=0.8pt] (\curX, 0) -- (\curX, 0.12);
    \node[below=3pt, lbl] at (\curX, 0) {\N};
  }

  % Sn Level bars
  \coordinate (S60) at ({\xR + 0.6 + 0*\dxR}, {\yzero + 3.29*\ym});
  \coordinate (S62) at ({\xR + 0.6 + 1*\dxR}, {\yzero + 1.22*\ym});
  \coordinate (S64) at ({\xR + 0.6 + 2*\dxR}, {\yzero + 0.12*\ym});
  \coordinate (S66) at ({\xR + 0.6 + 3*\dxR}, {\yzero + 0.13*\ym});
  \coordinate (S68) at ({\xR + 0.6 + 4*\dxR}, {\yzero + 0.92*\ym});
  \coordinate (S70) at ({\xR + 0.6 + 5*\dxR}, {\yzero + 2.28*\ym});
  \coordinate (S72) at ({\xR + 0.6 + 6*\dxR}, {\yzero + 4.16*\ym});
  \coordinate (S74) at ({\xR + 0.6 + 7*\dxR}, {\yzero + 6.80*\ym});
  \coordinate (S76) at ({\xR + 0.6 + 8*\dxR}, {\yzero + 9.91*\ym});

  \foreach \pt in {S60, S62, S64, S66, S68, S70, S72, S74, S76} {
    \draw[lvl] ($(\pt) - (\hw, 0)$) -- ($(\pt) + (\hw, 0)$);
  }

  % Dashed pairing rotational parabola
  \draw[conn, smooth] plot coordinates {(S60) (S62) (S64) (S66) (S68) (S70) (S72) (S74) (S76)};

  % Nucleus label
  \node[font=\large] at ({\xR + 1.8}, 2.4) {\textbf{Sn}};

\end{tikzpicture}
\end{document}
