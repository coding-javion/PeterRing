\documentclass[tikz,border=10pt]{standalone}
\usepackage{fontspec}
\usepackage[mathrm=sym]{unicode-math}
\setmainfont{texgyrepagella-regular.otf}[
    BoldFont = texgyrepagella-bold.otf,
    ItalicFont = texgyrepagella-italic.otf,
    BoldItalicFont = texgyrepagella-bolditalic.otf
]
\setmathfont{texgyrepagella-math.otf}[AutoFakeBold = 2.5]
\usepackage{tikz}
\usetikzlibrary{arrows.meta,patterns}

\begin{document}
\begin{tikzpicture}[>=Stealth, line cap=round, line join=round]

  % ========================================================
  % Panel 1: m = 0
  % ========================================================
  \begin{scope}[shift={(0,0)}]
    % z-axis (horizontal)
    \draw[thick, ->] (-2.2, 0) -- (2.4, 0) node[right=2pt, font=\Large] {$z$};

    % |Y_20|^2 pattern: lobes along horizontal (z) axis, small vertical lobe
    % Polar angle theta measured from horizontal (z) axis:
    % r(theta) = 1.6 * (3*cos(theta)^2 - 1)^2 / 4.0
    \draw[very thick, line width=1.2pt, pattern=north east lines, pattern color=black!75, domain=0:360, samples=180, smooth]
      plot (\x: {1.65 * (3 * (cos(\x))^2 - 1)^2 / 4.0});

    \node[font=\Large] at (0, -1.9) {$m = 0$};
  \end{scope}

  % ========================================================
  % Panel 2: m = +/- 1
  % ========================================================
  \begin{scope}[shift={(5.2,0)}]
    % z-axis (horizontal)
    \draw[thick, ->] (-2.0, 0) -- (2.4, 0) node[right=2pt, font=\Large] {$z$};

    % |Y_21|^2 pattern: r(theta) = 1.6 * sin(2*theta)^2
    \draw[very thick, line width=1.2pt, pattern=north east lines, pattern color=black!75, domain=0:360, samples=180, smooth]
      plot (\x: {1.65 * (sin(2*\x))^2});

    \node[font=\Large] at (0, -1.9) {$m = \pm 1$};
  \end{scope}

  % ========================================================
  % Panel 3: m = +/- 2
  % ========================================================
  \begin{scope}[shift={(10.4,0)}]
    % z-axis (horizontal)
    \draw[thick, ->] (-1.8, 0) -- (2.4, 0) node[right=2pt, font=\Large] {$z$};

    % |Y_22|^2 pattern: r(theta) = 1.6 * sin(theta)^4
    \draw[very thick, line width=1.2pt, pattern=north east lines, pattern color=black!75, domain=0:360, samples=180, smooth]
      plot (\x: {1.65 * (sin(\x))^4});

    \node[font=\Large] at (0, -1.9) {$m = \pm 2$};
  \end{scope}

\end{tikzpicture}
\end{document}
