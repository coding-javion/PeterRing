]

\usepackage{tikz}

\begin{document}
\begin{tikzpicture}[>=Stealth, line cap=round, line join=round,
  arrowed/.style={
    decoration={
      markings,
      mark=at position 0.55 with {\arrow{Stealth[length=5pt, width=3.5pt]}}
    },
    postaction={decorate}
  }
]

  % Left vertical particle line split at interaction vertex (-1.8, 0.9)
  \draw[thick, line width=1.1pt, arrowed] (-1.8, -1.8) -- (-1.8, 0.9);
  \draw[thick, line width=1.1pt, arrowed] (-1.8, 0.9) -- (-1.8, 1.8);

  % Right vertical particle line split at interaction vertex (1.8, -0.9)
  \draw[thick, line width=1.1pt, arrowed] (1.8, -1.8) -- (1.8, -0.9);
  \draw[thick, line width=1.1pt, arrowed] (1.8, -0.9) -- (1.8, 1.8);

  % Top dashed interaction line
  \draw[thick, dashed, line width=1pt] (-1.8, 0.9) -- (0, 0.9);

  % Bottom dashed interaction line
  \draw[thick, dashed, line width=1pt] (0, -0.9) -- (1.8, -0.9);

  % Particle-hole bubble in the middle
  \coordinate (VT) at (0, 0.9);
  \coordinate (VB) at (0, -0.9);

  % Left arc of bubble: pointing down (hole)
  \draw[thick, line width=1.1pt, arrowed] (VT) to[out=-150, in=150] (VB);

  % Right arc of bubble: pointing up (particle)
  \draw[thick, line width=1.1pt, arrowed] (VB) to[out=30, in=-30] (VT);

  % Vertices at bubble ends and interaction line ends
  \filldraw[black] (VT) circle (2pt);
  \filldraw[black] (VB) circle (2pt);
  \filldraw[black] (-1.8, 0.9) circle (2pt);
  \filldraw[black] (1.8, -0.9) circle (2pt);

\end{tikzpicture}
\end{document}
