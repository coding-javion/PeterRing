\begin{tikzpicture}[>=Stealth, line cap=round, line join=round]

  % Potential well curve
  % Flat outside, smooth Woods-Saxon-like profile
  \draw[line width=1.1pt] 
    (-4.5, 0) -- (-2.3, 0)
    .. controls (-1.8, 0) and (-1.7, -3.6) .. (0, -3.6)
    .. controls (1.7, -3.6) and (1.8, 0) .. (2.3, 0)
    -- (4.8, 0);

  % Central vertical axis (Energy)
  \draw[line width=0.85pt, ->] (0, -4.3) -- (0, 3.8);
  \node[rotate=90, font=\large, anchor=south] at (-0.08, 2.0) {Energy};

  % Energy levels inside well
  % 1s level
  \draw[line width=0.75pt] (-0.85, -3.05) -- (0.85, -3.05);
  \node[left=4pt, font=\normalsize] at (-0.95, -3.05) {$1s$};

  % 1p level
  \draw[line width=0.75pt] (-1.35, -1.95) -- (1.35, -1.95);
  \node[left=4pt, font=\normalsize] at (-1.35, -1.95) {$1p$};

  % 2s, 1d level
  \draw[line width=0.75pt] (-1.65, -1.0) -- (1.65, -1.0);
  \node[left=4pt, font=\normalsize] at (-1.65, -1.0) {$2s, 1d$};

  % Hole on 1p level (open circle)
  \draw[line width=0.9pt, fill=white] (0.55, -1.95) circle (2.2pt);

  % Dashed line extending from 1p level to the right
  \draw[line width=0.75pt, dashed] (1.35, -1.95) -- (3.5, -1.95);

  % Separation energy E_B (double-headed arrow)
  \draw[line width=0.75pt, <->] (2.45, 0) -- (2.45, -1.95)
    node[midway, fill=white, inner sep=1.5pt, font=\normalsize] {$E_B$};

  % Incident proton E (top left)
  \fill (-4.1, 3.2) circle (2.0pt);
  \draw[line width=0.9pt, ->] (-4.1, 3.2) -- (-1.6, 3.2)
    node[midway, above=1pt, font=\normalsize] {$E$};

  % Outgoing protons E' and E'' (top right)
  \fill (2.2, 1.7) circle (2.0pt);
  \draw[line width=0.9pt, ->] (2.2, 1.7) -- (4.8, 1.7)
    node[midway, above=1pt, font=\normalsize] {$E'$};

  \fill (2.2, 0.9) circle (2.0pt);
  \draw[line width=0.9pt, ->] (2.2, 0.9) -- (4.8, 0.9)
    node[midway, above=1pt, font=\normalsize] {$E''$};

\end{tikzpicture}
