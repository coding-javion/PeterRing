\documentclass[tikz,border=10pt]{standalone}
\usepackage{fontspec}
\usepackage[mathrm=sym]{unicode-math}
\setmainfont{texgyrepagella-regular.otf}[
    BoldFont = texgyrepagella-bold.otf,
    ItalicFont = texgyrepagella-italic.otf,
    BoldItalicFont = texgyrepagella-bolditalic.otf
]
\setmathfont{texgyrepagella-math.otf}[AutoFakeBold = 2.5]
\usepackage{tikz}
\usetikzlibrary{arrows.meta}

\begin{document}
\begin{tikzpicture}[>=Stealth, line cap=round, line join=round]

  % Scale definition
  \def\w{4.8}
  \def\h{3.6}
  \def\yOne{2.6}
  \def\epsK{3.0}

  % Precalculate y at epsK: lambda = 2.5, Delta = 0.65
  % 0.5 * 2.6 * (1 - (3.0 - 2.5)/sqrt((3.0 - 2.5)^2 + 0.65^2))
  % = 1.3 * (1 - 0.5 / sqrt(0.25 + 0.4225)) = 1.3 * (1 - 0.5/0.820) = 1.3 * 0.390 = 0.507
  \def\yValK{0.51}

  % ========================================================
  % Left Panel: Unblocked Fermi distribution with odd nucleon at eps_k1
  % ========================================================
  \begin{scope}[shift={(0,0)}]
    % Axes
    \draw[thick, ->] (-0.2, 0) -- (\w, 0) node[below=2pt, font=\Large] {$\varepsilon_k$};
    \draw[thick, ->] (0, -0.2) -- (0, \h) node[left=2pt, font=\Large] {$v_k^2$};

    % Tick at 1
    \draw[thick] (-0.08, \yOne) -- (0.08, \yOne) node[left=4pt, font=\large] {$1$};

    % BCS smooth curve: v_k^2
    \draw[very thick, line width=1.4pt, smooth, domain=0:4.3, samples=120]
      plot (\x, {0.5 * \yOne * (1.0 - (\x - 2.5) / sqrt((\x - 2.5)^2 + 0.65^2))});

    % Dashed vertical line to curve at eps_k1
    \draw[thick, dashed] (\epsK, 0) -- (\epsK, \yValK);

    % Solid vertical spike at eps_k1 rising to 1
    \draw[very thick, line width=1.4pt] (\epsK, \yValK) -- (\epsK, \yOne);

    % Tick and label at eps_k1
    \draw[thick] (\epsK, -0.08) -- (\epsK, 0.08);
    \node[below=4pt, font=\Large] at (\epsK, 0) {$\varepsilon_{k_1}$};
  \end{scope}

  % ========================================================
  % Right Panel: Blocked Fermi distribution with gap at eps_k1
  % ========================================================
  \begin{scope}[shift={(6.2,0)}]
    % Axes
    \draw[thick, ->] (-0.2, 0) -- (\w, 0) node[below=2pt, font=\Large] {$\varepsilon_k$};
    \draw[thick, ->] (0, -0.2) -- (0, \h) node[left=2pt, font=\Large] {$v_{\bar{k}}^2$};

    % Tick at 1
    \draw[thick] (-0.08, \yOne) -- (0.08, \yOne) node[left=4pt, font=\large] {$1$};

    % Gap width around eps_k1
    \def\gap{0.07}
    \def\yLeft{0.56}
    \def\yRight{0.46}

    % BCS smooth curve before eps_k1
    \draw[very thick, line width=1.4pt, smooth, domain=0:{3.0 - \gap}, samples=80]
      plot (\x, {0.5 * \yOne * (1.0 - (\x - 2.5) / sqrt((\x - 2.5)^2 + 0.65^2))});

    % Vertical cut line at eps_k1 - gap
    \draw[thick] ({3.0 - \gap}, 0) -- ({3.0 - \gap}, \yLeft);

    % Vertical cut line at eps_k1 + gap
    \draw[thick] ({3.0 + \gap}, 0) -- ({3.0 + \gap}, \yRight);

    % BCS smooth curve after eps_k1
    \draw[very thick, line width=1.4pt, smooth, domain={3.0 + \gap}:4.3, samples=60]
      plot (\x, {0.5 * \yOne * (1.0 - (\x - 2.5) / sqrt((\x - 2.5)^2 + 0.65^2))});

    % Tick and label at eps_k1
    \draw[thick] (3.0, -0.08) -- (3.0, 0.08);
    \node[below=4pt, font=\Large] at (3.0, 0) {$\varepsilon_{k_1}$};
  \end{scope}

\end{tikzpicture}
\end{document}
