  arrowed/.style={

  % LHS equation symbol
  \node[font=\large] at (-2.2, 0) {$M^{(2)} \equiv$};

  \coordinate (VL) at (-1.2, 0);
  \coordinate (VR) at (1.2, 0);

  % Top curved propagator
  \draw[thick, line width=1.1pt, arrowed] (VL) to[out=60, in=120] (VR);

  % Middle straight propagator
  \draw[thick, line width=1.1pt, arrowed] (VL) -- (VR);

  % Bottom curved propagator
  \draw[thick, line width=1.1pt, arrowed] (VL) to[out=-60, in=-120] (VR);

  % Vertices
  \filldraw[black] (VL) circle (2.2pt);
  \filldraw[black] (VR) circle (2.2pt);
