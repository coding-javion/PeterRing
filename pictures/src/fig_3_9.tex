    \pgfmathsetmacro{\y}{\yzero + #3 * \yscale}
    \draw[line width=1.2pt] (#1, \y) node[left=1.5pt, font=\footnotesize] {$#4$} -- (#2, \y);%
  }

    \pgfmathsetmacro{\y}{\yzero + #3 * \yscale}
    \draw[line width=1.2pt] (#1, \y) -- (#2, \y) node[right=1.5pt, font=\footnotesize] {$#4$};%
