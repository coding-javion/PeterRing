  % Potential well boundary
  \draw[line width=1.4pt]
    (-3.4, 0.9) to[out=0, in=125] (-2.6, 0.3)
    to[out=-55, in=100] (-2.1, -1.0)
    to[out=-80, in=180] (-1.3, -2.0)
    -- (1.3, -2.0)
    to[out=0, in=-100] (2.1, -1.0)
    to[out=80, in=-125] (2.6, 0.3)
    to[out=55, in=180] (3.4, 0.9);

  % Horizontal dividing line (Fermi surface)
  \draw[line width=1.1pt] (-2.28, 0) -- (2.28, 0);

  % 5 Particles (crosses) above the line at y = 0.7
  \foreach \x in {-1.4, -0.7, 0.0, 0.7, 1.4} {
    \draw[line width=1.4pt] ({\x - 0.12}, {0.7 - 0.12}) -- ({\x + 0.12}, {0.7 + 0.12});
    \draw[line width=1.4pt] ({\x - 0.12}, {0.7 + 0.12}) -- ({\x + 0.12}, {0.7 - 0.12});
  }

  % 5 Holes (open circles) below the line at y = -0.7
  \foreach \x in {-1.4, -0.7, 0.0, 0.7, 1.4} {
    \draw[line width=1.4pt] (\x, -0.7) circle (3.2pt);
  }

\end{tikzpicture}
